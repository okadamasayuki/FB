import tempfile
import traceback
from collections.abc import Generator
from pathlib import Path
from typing import Any

from dify_plugin import Tool
from dify_plugin.entities.tool import ToolInvokeMessage

from core_logic import notebook, python_runner


class RunPythonTool(Tool):
    def _invoke(self, tool_parameters: dict[str, Any]) -> Generator[ToolInvokeMessage, None, None]:
        try:
            yield from self._run(tool_parameters)
        except Exception:
            yield self.create_text_message(
                "予期しないエラーが発生しました:\n" + traceback.format_exc()
            )
            yield self.create_json_message({"status": "error", "message": "unexpected error"})

    def _run(self, tool_parameters: dict[str, Any]) -> Generator[ToolInvokeMessage, None, None]:
        code = (tool_parameters.get("code") or "").strip()
        if not code:
            yield self.create_text_message(
                "実行するプログラムが設定されていません。\n"
                "ワークフローの環境変数 `PYTHON_CODE` に Python プログラムを貼り付けてから、"
                "もう一度公開してください。"
            )
            yield self.create_json_message({"status": "error", "message": "no code"})
            return

        files = tool_parameters.get("files") or []
        attachments = [
            (getattr(f, "filename", None) or f"file{i + 1}", f.blob) for i, f in enumerate(files)
        ]
        user_text = tool_parameters.get("user_text") or ""
        title = (tool_parameters.get("title") or "").strip() or "Python 実行"

        with tempfile.TemporaryDirectory() as tmp:
            result = python_runner.run_python(code, attachments, Path(tmp), user_input=user_text)

            # 必要なファイルが足りない → 実行せずに聞き返す
            if result.missing:
                lines = ["このプログラムの実行には、次のファイルの添付が必要です。", ""]
                lines += [f"- **{name}**" for name in result.missing]
                lines += [
                    "",
                    "ファイルを添付して、もう一度送信してください。",
                ]
                if result.attached:
                    lines.append(f"(添付済み: {', '.join(result.attached)})")
                yield self.create_text_message("\n".join(lines))
                yield self.create_json_message(
                    {"status": "need_files", "missing": result.missing, "attached": result.attached}
                )
                return

            output_names = [p.name for p in result.output_files]
            yield self.create_text_message(
                notebook.python_run_markdown(
                    title, result.success, result.log, output_names, result.attached
                )
            )
            report_html = notebook.python_run_html(
                title, result.success, code, result.log, output_names, result.attached
            )
            yield self.create_blob_message(
                blob=report_html.encode("utf-8"),
                meta={"mime_type": notebook.HTML_MIME, "filename": f"実行レポート_{title}.html"},
            )
            for path in result.output_files:
                yield self.create_blob_message(
                    blob=path.read_bytes(),
                    meta={"mime_type": notebook.mime_for(path.name), "filename": path.name},
                )
            yield self.create_json_message(
                {
                    "status": "success" if result.success else "error",
                    "attached": result.attached,
                    "output_files": output_names,
                }
            )

"""貼り付けられた Python を実行する土台。

資金計画のコア実行(core_runner)と、汎用の Python 実行(python_runner)の
どちらもここを使う。資金計画に依存しないので、Python 実行だけのプラグインにも
そのまま入る。
"""

from __future__ import annotations

import contextlib
import inspect
import io
import traceback
import types
from dataclasses import dataclass, field
from pathlib import Path


@dataclass
class CoreResult:
    success: bool
    log: str
    output_files: list[Path] = field(default_factory=list)


def execute_code(
    code_text: str,
    variables: dict,
    output_dir: str | Path,
    error_label: str = "貼り付けた Python",
) -> CoreResult:
    """貼り付けられた Python を、変数を与えて実行する共通処理。

    - variables はコードの中で最初から使えるグローバル変数になる
    - モジュールとして実行するため __name__ は "__main__" ではなく、
      `if __name__ == "__main__":` のブロックは実行されない(main() は下で呼ぶ)
    - def main(...) が定義されていれば、定義にある引数だけを渡して呼ぶ
    - 出力ファイルは main の戻り値、無ければ output_dir から自動回収する
    """
    buffer = io.StringIO()
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)
    module = types.ModuleType("core_injected")
    module.__dict__.update(variables)
    try:
        compiled = compile(code_text, f"<{error_label}>", "exec")
        with contextlib.redirect_stdout(buffer), contextlib.redirect_stderr(buffer):
            exec(compiled, module.__dict__)
            main = module.__dict__.get("main")
            outputs = main(**accepted_kwargs(main, variables)) if callable(main) else None
        return collect(buffer, outputs, output_dir)
    except Exception:
        log = buffer.getvalue() + f"\n{error_label} でエラーが発生しました:\n" + traceback.format_exc()
        return CoreResult(success=False, log=log)


def accepted_kwargs(func, params: dict) -> dict:
    """main() の定義にある引数だけを渡す(貼り付け側の書き方の自由度のため)。"""
    try:
        sig = inspect.signature(func)
    except (TypeError, ValueError):
        return {}
    if any(p.kind is inspect.Parameter.VAR_KEYWORD for p in sig.parameters.values()):
        return dict(params)
    return {name: value for name, value in params.items() if name in sig.parameters}


def collect(buffer: io.StringIO, outputs, scan_dir: Path | None) -> CoreResult:
    """出力ファイルを確定する。outputs(main の戻り値)が無ければ scan_dir から拾う。"""
    if outputs is None and scan_dir is not None:
        files = sorted(p for p in scan_dir.rglob("*") if p.is_file())
    else:
        files = [Path(p) for p in outputs or []]
        missing = [p for p in files if not p.exists()]
        if missing:
            names = ", ".join(p.name for p in missing)
            return CoreResult(
                success=False,
                log=buffer.getvalue() + f"\nエラー: 出力ファイルが見つかりません: {names}",
            )
    log = buffer.getvalue()
    if not files:
        log += "\n注意: 出力ファイルが作られませんでした(output_dir に保存してください)。"
    return CoreResult(success=True, log=log, output_files=files)

"""汎用の Python 実行(Jupyter Lab の代替)。

Jupyter Lab が使えない環境で、Dify を Python の実行環境として使うための仕組み。
プログラムはワークフローの環境変数(PYTHON_CODE)に貼り付けておき、
チャットに添付されたファイルを入力として実行する。

プログラム側の約束事:

- 先頭付近のコメントで必要な入力ファイルを宣言できる(無くてもよい)
      # 入力ファイル: 元データ.xlsm, マスタ.csv
  宣言があるのに添付が足りないときは実行せず、「このファイルを添付してください」と
  返す(チャットで聞き返す動きになる)。
- コードの中では次の変数が最初から使える:
      input_dir    添付ファイルが置かれたフォルダ(Path)
      input_files  添付ファイルのパスのリスト(添付順)
      input_path   最初の添付ファイル(無ければ None)
      output_dir   出力ファイルの保存先(ここに保存すれば自動で回収される)
      user_input   チャットで一緒に入力されたテキスト
- print() はそのまま実行レポートに表示される。
- def main(...) を定義する書き方でも動く(core_runner.execute_code と同じ規約)。
"""

from __future__ import annotations

import re
from dataclasses import dataclass, field
from pathlib import Path

from .runner import execute_code

RE_REQUIRED = re.compile(r"^#\s*(?:入力|必要)ファイル\s*[::]\s*(.+)$", re.MULTILINE)


def required_files(code: str) -> list[str]:
    """コード先頭のコメントで宣言された入力ファイル名の一覧を返す。"""
    names: list[str] = []
    for m in RE_REQUIRED.finditer(str(code)):
        for part in re.split(r"[,、]", m.group(1)):
            name = part.strip()
            if name and name not in names:
                names.append(name)
    return names


@dataclass
class PyRunResult:
    success: bool
    log: str = ""
    output_files: list[Path] = field(default_factory=list)
    required: list[str] = field(default_factory=list)  # 宣言された入力ファイル
    missing: list[str] = field(default_factory=list)  # 添付されていないもの
    attached: list[str] = field(default_factory=list)  # 実際に添付されたファイル名


def _is_attached(declared: str, attached_names: list[str]) -> bool:
    """宣言名と添付名の照合。完全一致のほか、部分一致も許す。

    (「元データ.xlsm」と宣言して「元データ_R7.xlsm」を添付した、
     宣言は「元データ」だけ、のようなズレを吸収するため)
    """
    for name in attached_names:
        if declared == name or declared in name or name in declared:
            return True
    return False


def run_python(
    code: str,
    attachments: list[tuple[str, bytes]],
    work_dir: str | Path,
    user_input: str = "",
) -> PyRunResult:
    """貼り付けられたプログラムを、添付ファイルを入力として実行する。"""
    work_dir = Path(work_dir)
    input_dir = work_dir / "input"
    output_dir = work_dir / "output"
    input_dir.mkdir(parents=True, exist_ok=True)

    input_files: list[Path] = []
    for name, blob in attachments:
        path = input_dir / (Path(name).name or "file")
        path.write_bytes(blob)
        input_files.append(path)
    attached_names = [p.name for p in input_files]

    required = required_files(code)
    missing = [name for name in required if not _is_attached(name, attached_names)]
    if missing:
        return PyRunResult(
            success=False, required=required, missing=missing, attached=attached_names
        )

    variables = {
        "input_dir": input_dir,
        "input_files": input_files,
        "input_path": input_files[0] if input_files else None,
        "output_dir": output_dir,
        "user_input": user_input,
    }
    result = execute_code(code, variables, output_dir, error_label="貼り付けた Python プログラム")
    return PyRunResult(
        success=result.success,
        log=result.log,
        output_files=result.output_files,
        required=required,
        attached=attached_names,
    )

"""Jupyter Lab 風のレポート表示。資金計画にも汎用 Python 実行にも使う共通部分。

ここは資金計画に依存しないので、Python 実行だけのプラグインにもそのまま入る。
"""

from __future__ import annotations

import html
from pathlib import Path

HTML_MIME = "text/html"

def mime_for(name: str) -> str:
    """ファイル名の拡張子から MIME タイプを決める(ダウンロード用)。"""
    suffix = Path(name).suffix.lower()
    return {
        ".xlsx": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        ".xlsm": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        ".csv": "text/csv",
        ".txt": "text/plain",
        ".html": "text/html",
        ".json": "application/json",
        ".pdf": "application/pdf",
        ".png": "image/png",
        ".jpg": "image/jpeg",
    }.get(suffix, "application/octet-stream")


def python_run_markdown(title: str, success: bool, log: str, output_files: list[str],
                        attached: list[str]) -> str:
    """汎用 Python 実行のチャット返信用テキスト。"""
    lines = [f"## {title}", ""]
    if attached:
        lines.append("- 入力ファイル: " + ", ".join(attached))
    lines.append("- 結果: " + ("実行成功" if success else "**エラーあり**(下のログ参照)"))
    lines += ["", "### print 出力", "", "```", log.rstrip("\n") or "(出力なし)", "```"]
    if output_files:
        lines += ["", "### 出力ファイル", ""]
        lines += [f"- {name}" for name in output_files]
    return "\n".join(lines)


def python_run_html(title: str, success: bool, code: str, log: str,
                    output_files: list[str], attached: list[str]) -> str:
    """汎用 Python 実行の Jupyter Lab 風レポート(ダウンロード用 HTML)。"""
    status = (
        '<span class="badge ok">実行成功</span>'
        if success
        else '<span class="badge ng">エラーあり</span>'

        rows = "<tr><th>入力ファイル</th><td>" + html.escape(", ".join(attached)) + "</td></tr>"
