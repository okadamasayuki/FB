import traceback
from collections.abc import Generator
from pathlib import Path
from typing import Any

from dify_plugin import Tool
from dify_plugin.entities.tool import ToolInvokeMessage

from core_logic import core_runner, preprocess, procedure, report
from core_logic.extract_config import ConfigError, parse_config
from core_logic.preprocess import XLSX_MIME, PreprocessError
from core_logic.procedure import ProcedureError


class GeneratePlanTool(Tool):
    def _invoke(self, tool_parameters: dict[str, Any]) -> Generator[ToolInvokeMessage, None, None]:
        try:
            yield from self._run(tool_parameters)
        except (ConfigError, PreprocessError, ProcedureError) as e:
            # 利用者の入力(手順書・設定 JSON・元データ)に起因するエラー
            yield self.create_text_message(f"エラー:\n{e}")
            yield self.create_json_message({"status": "error", "message": str(e)})
        except Exception:
            yield self.create_text_message(
                "予期しないエラーが発生しました:\n" + traceback.format_exc()
            )
            yield self.create_json_message({"status": "error", "message": "unexpected error"})

    def _run(self, tool_parameters: dict[str, Any]) -> Generator[ToolInvokeMessage, None, None]:
        source = tool_parameters.get("source_file")
        if source is None:
            raise ProcedureError("元データ Excel がアップロードされていません。")

        procedure_text, procedure_name = self._read_procedure(tool_parameters)
        config_json = (tool_parameters.get("config_json") or "").strip()
        if not procedure_text and not config_json:
            raise ProcedureError(
                "手順書ファイル(または手順書の本文)を指定してください。\n"
                "Excel 抜粋エディタの「手順書」ボタンから書き出したテキストファイルを使います。"
            )

        fiscal_year = tool_parameters.get("fiscal_year")
        fiscal_year = int(fiscal_year) if fiscal_year else None
        core_code = (tool_parameters.get("core_code") or "").strip()

        with tempfile.TemporaryDirectory() as tmp:
            tmp_dir = Path(tmp)
            source_name = getattr(source, "filename", None) or "source.xlsm"
            source_path = tmp_dir / source_name
            source_path.write_bytes(source.blob)

            if procedure_text:
                run_report, bulk_path = self._run_procedure(
                    source_path, source_name, procedure_text, procedure_name,
                    fiscal_year, core_code, tmp_dir,
                )
            else:
                run_report, bulk_path = self._run_config(
                    source_path, source_name, config_json, fiscal_year, core_code, tmp_dir
                )

            yield self.create_text_message(report.to_markdown(run_report))

            # Jupyter Lab 風の実行レポート(ブラウザで開ける HTML)
            report_path = report.write_html(
                run_report, tmp_dir / f"実行レポート_{run_report.plan_fiscal_year}年度.html"
            )
            yield self.create_blob_message(
                blob=report_path.read_bytes(),
                meta={"mime_type": report.HTML_MIME, "filename": report_path.name},
            )
            yield self.create_blob_message(
                blob=bulk_path.read_bytes(),
                meta={"mime_type": XLSX_MIME, "filename": bulk_path.name},
            )
            for name in run_report.output_files:
                path = tmp_dir / "output" / name
                yield self.create_blob_message(
                    blob=path.read_bytes(),
                    meta={"mime_type": _mime_for(name), "filename": name},
                )
            yield self.create_json_message(
                {
                    "status": "success" if run_report.success else "error",
                    "mode": "procedure" if procedure_text else "config_json",
                    "core": "injected" if core_code else "bundled_mock",
                    "plan_fiscal_year": run_report.plan_fiscal_year,
                    "bulk_file": bulk_path.name,
                    "output_files": run_report.output_files,
                    "warnings": run_report.warnings,
                }
            )

    def _read_procedure(self, tool_parameters: dict[str, Any]) -> tuple[str, str]:
        """手順書をファイルまたは本文から読む。戻り値は (本文, ファイル名)。"""
        text = (tool_parameters.get("procedure_text") or "").strip()
        name = ""
        pf = tool_parameters.get("procedure_file")
        if pf is not None:
            name = getattr(pf, "filename", None) or "手順書.txt"
            blob = pf.blob
            for encoding in ("utf-8-sig", "utf-8", "cp932"):
                try:
                    text = blob.decode(encoding).strip()
                    break
                except UnicodeDecodeError:
                    continue
            else:
                raise ProcedureError(
                    f"手順書ファイル({name})の文字コードを判別できません。"
                    "UTF-8 で保存し直してください。"
                )
        return text, name

    def _run_procedure(
        self,
        source_path: Path,
        source_name: str,
        procedure_text: str,
        procedure_name: str,
        fiscal_year: int | None,
        core_code: str,
        tmp_dir: Path,
    ):
        if not fiscal_year:
            raise ProcedureError(
                "「実績データの年度」を入力してください(例: 2025)。\n"
                "手順書には年度の情報が入らないため、ここで指定する必要があります。"
            )
        plan_fiscal_year = fiscal_year + 1

        sheets = procedure.load_source(source_path)
        result = procedure.run_script(procedure_text, sheets)
        bulk_path = procedure.write_output(
            result, sheets, tmp_dir / f"入力用一括_{fiscal_year}年度.xlsx"
        )

        core = core_runner.run_core(
            bulk_path, tmp_dir / "output", fiscal_year, plan_fiscal_year, core_code=core_code
        )
        return (
            report.from_procedure(
                result=result,
                core_log=core.log,
                success=core.success and not result.errors,
                bulk_file=bulk_path.name,
                fiscal_year=fiscal_year,
                plan_fiscal_year=plan_fiscal_year,
                output_files=[p.name for p in core.output_files],
                source_file=source_name,
                procedure_file=procedure_name,
                core_label=_core_label(core_code),
                core_code=core_code,
            ),
            bulk_path,
        )

    def _run_config(
        self,
        source_path: Path,
        source_name: str,
        config_json: str,
        fiscal_year: int | None,
        core_code: str,
        tmp_dir: Path,
    ):
        """旧方式(設定 JSON で行・列を指定)。手順書へ移行するまでの互換用。"""
        config = parse_config(config_json, fiscal_year)
        bulk_path = tmp_dir / f"入力用一括_{config.fiscal_year}年度.xlsx"
        extract, _ = preprocess.build_bulk_excel(source_path, config, bulk_path)
        core = core_runner.run_core(
            bulk_path,
            tmp_dir / "output",
            config.fiscal_year,
            config.plan_fiscal_year,
            config.start_month,
            core_code=core_code,
        )
        return (
            report.from_extract(
                extract=extract,
                core_log=core.log,
                success=core.success,
                bulk_file=bulk_path.name,
                output_files=[p.name for p in core.output_files],
                source_file=source_name,
                core_label=_core_label(core_code),
                core_code=core_code,
            ),
            bulk_path,
        )

    if core_code:

    return report.mime_for(name)

import tempfile
import traceback
from collections.abc import Generator
from pathlib import Path
from typing import Any

from dify_plugin import Tool
from dify_plugin.entities.tool import ToolInvokeMessage

from core_logic import python_runner, report


class RunPythonTool(Tool):
    def _invoke(self, tool_parameters: dict[str, Any]) -> Generator[ToolInvokeMessage, None, None]:
        try:
            yield from self._run(tool_parameters)
        except Exception:
            yield self.create_text_message(
                "予期しないエラーが発生しました:\n" + traceback.format_exc()
            )
            yield self.create_json_message({"status": "error", "message": "unexpected error"})

    def _run(self, tool_parameters: dict[str, Any]) -> Generator[ToolInvokeMessage, None, None]:
        code = (tool_parameters.get("code") or "").strip()
        if not code:
            yield self.create_text_message(
                "実行するプログラムが設定されていません。\n"
                "ワークフローの環境変数 `PYTHON_CODE` に Python プログラムを貼り付けてから、"
                "もう一度公開してください。"
            )
            yield self.create_json_message({"status": "error", "message": "no code"})
            return

        files = tool_parameters.get("files") or []
        attachments = [
            (getattr(f, "filename", None) or f"file{i + 1}", f.blob) for i, f in enumerate(files)
        ]
        user_text = tool_parameters.get("user_text") or ""
        title = (tool_parameters.get("title") or "").strip() or "Python 実行"

        with tempfile.TemporaryDirectory() as tmp:
            result = python_runner.run_python(code, attachments, Path(tmp), user_input=user_text)

            # 必要なファイルが足りない → 実行せずに聞き返す
            if result.missing:
                lines = ["このプログラムの実行には、次のファイルの添付が必要です。", ""]
                lines += [f"- **{name}**" for name in result.missing]
                lines += [
                    "",
                    "ファイルを添付して、もう一度送信してください。",
                ]
                if result.attached:
                    lines.append(f"(添付済み: {', '.join(result.attached)})")
                yield self.create_text_message("\n".join(lines))
                yield self.create_json_message(
                    {"status": "need_files", "missing": result.missing, "attached": result.attached}
                )
                return

            output_names = [p.name for p in result.output_files]
            yield self.create_text_message(
                report.python_run_markdown(
                    title, result.success, result.log, output_names, result.attached
                )
            )
            report_html = report.python_run_html(
                title, result.success, code, result.log, output_names, result.attached
            )
            yield self.create_blob_message(
                blob=report_html.encode("utf-8"),
                meta={"mime_type": report.HTML_MIME, "filename": f"実行レポート_{title}.html"},
            )
            for path in result.output_files:
                yield self.create_blob_message(
                    blob=path.read_bytes(),
                    meta={"mime_type": report.mime_for(path.name), "filename": path.name},
                )
            yield self.create_json_message(
                {
                    "status": "success" if result.success else "error",
                    "attached": result.attached,
                    "output_files": output_names,
                }
            )

"""Excel 抜粋エディタの「手順書」を Python で実行する。

Excel 抜粋エディタ(https://github.com/okadamasayuki/excel-editor)は
ブラウザで動く JavaScript のツールで、ドラッグ&ドロップで組んだ抜き出し操作を
テキストの「手順書」として書き出せる。

この モジュールは、その手順書を **同じ意味で** 解釈して openpyxl で実行するもの。
元の実装(src/index.html の expandScript / execLine / runScript)を移植している。

対応している書き方:

    # コメント / 説明: … / 元ファイル: …      … 読み飛ばす
    シート追加 <名前>                          … 出力シートを用意(同名は使い回し)
    <シート>の<範囲>を<出力シート>の<セル>に置く
    [<ファイル名>]<シート>の…                 … 同名シートの絞り込み
    …を続けて置く                              … 直前の下に続ける
    …を右に続けて置く                          … 直前の右に続ける
    …に置く(転置)                              … 行と列を入れ替える
    繰り返し <変数> = <値,…> 〜 ここまで       … 値の数だけ繰り返し、{変数} を差し替え
    生成                                       … そこまでの結果を書き出す

範囲の書き方: A1からD10 / A1:D10 / 3行目 / 2行目から10行目 / B列からD列 / B列 / 全体
"""

import re
from dataclasses import dataclass, field
from pathlib import Path

from openpyxl import Workbook, load_workbook
from openpyxl.utils import get_column_letter

MAX_COLS = 16384


class ProcedureError(ValueError):
    """手順書の内容に問題があるときに送出する。message は利用者向けの日本語。"""


# --- 文字列・座標のユーティリティ(元実装の toHalf / colName / colIndex 相当) ---


def to_half(text: object) -> str:
    """全角→半角。日本語入力そのままの手順書を受け取れるようにする。"""
    out = []
    for ch in str(text):
        code = ord(ch)
        if 0xFF01 <= code <= 0xFF5E:  # ！-～
            out.append(chr(code - 0xFEE0))
        elif ch == "　":  # 全角スペース
            out.append(" ")
        else:
            out.append(ch)
    return "".join(out)


def col_name(index: int) -> str:
    """0 始まりの列番号 → A, B, …, AA。"""
    name = ""
    n = index + 1
    while n > 0:
        n, rem = divmod(n - 1, 26)
        name = chr(65 + rem) + name
    return name


def col_index(letters: str) -> int:
    """A, B, …, AA → 0 始まりの列番号。読めなければ -1。"""
    n = 0
    for ch in str(letters).upper():
        v = ord(ch) - 64
        if v < 1 or v > 26:
            return -1
        n = n * 26 + v
    return n - 1


def parse_cell(text: str) -> tuple[int, int] | None:
    """"B3" → (行, 列) の 0 始まり。"""
    m = re.fullmatch(r"\$?([A-Za-z]{1,3})\$?(\d{1,7})", str(text or "").strip())
    if not m:
        return None
    c, r = col_index(m.group(1)), int(m.group(2)) - 1
    if c < 0 or r < 0:
        return None
    return r, c


def a1(row: int, col: int) -> str:
    return f"{col_name(col)}{row + 1}"


@dataclass(frozen=True)
class Range:
    r1: int
    c1: int
    r2: int
    c2: int

    @property
    def height(self) -> int:
        return self.r2 - self.r1 + 1

    @property
    def width(self) -> int:
        return self.c2 - self.c1 + 1

    def label(self) -> str:
        return f"{a1(self.r1, self.c1)}:{a1(self.r2, self.c2)}"


def norm_range(r1: int, c1: int, r2: int, c2: int) -> Range:
    return Range(min(r1, r2), min(c1, c2), max(r1, r2), max(c1, c2))


# --- 元データ ---


@dataclass
class SourceSheet:
    name: str
    file: str
    values: list[list[object]]  # [行][列]

    @property
    def used_rows(self) -> int:
        return len(self.values)

    @property
    def used_cols(self) -> int:
        return max((len(r) for r in self.values), default=0)

    def value_at(self, row: int, col: int) -> object:
        if 0 <= row < len(self.values) and 0 <= col < len(self.values[row]):
            return self.values[row][col]
        return None


def load_source(paths: str | Path | list[str | Path]) -> list[SourceSheet]:
    """元データ Excel(複数可)を読み込む。マクロは実行せず値だけを読む。"""
    if not isinstance(paths, list):
        paths = [paths]
    sheets: list[SourceSheet] = []
    for path in paths:
        path = Path(path)
        if not path.exists():
            raise ProcedureError(f"元データファイルが見つかりません: {path}")
        try:
            workbook = load_workbook(path, data_only=True)
        except Exception as e:
            raise ProcedureError(
                f"元データを Excel として開けません({path.name})。詳細: {e}"
            ) from e
        try:
            for ws in workbook.worksheets:
                rows = [list(row) for row in ws.iter_rows(values_only=True)]
                sheets.append(SourceSheet(name=ws.title, file=path.name, values=rows))
        finally:
            workbook.close()
    return sheets


# --- 出力(ブロックの集まり) ---


@dataclass
class Block:
    sheet: int  # 元データのシート番号
    src: Range
    dr: int  # 置き先の行(0 始まり)
    dc: int  # 置き先の列(0 始まり)
    trans: bool = False

    @property
    def height(self) -> int:
        return self.src.width if self.trans else self.src.height

    @property
    def width(self) -> int:
        return self.src.height if self.trans else self.src.width


@dataclass
class OutSheet:
    name: str
    blocks: list[Block] = field(default_factory=list)

    def max_dest_row(self) -> int:
        """使われている最終行(何も無ければ -1)。"""
        m = -1
        for b in self.blocks:
            m = max(m, b.dr + b.height - 1)
        return m

    def max_dest_col(self) -> int:
        m = -1
        for b in self.blocks:
            m = max(m, b.dc + b.width - 1)
        return m


@dataclass
class RunResult:
    out_sheets: list[OutSheet]
    applied: int = 0
    errors: list[dict] = field(default_factory=list)
    messages: list[str] = field(default_factory=list)
    generate: bool = False

    def summary_lines(self) -> list[str]:
        lines = [f"手順書どおりに {self.applied} 件を配置しました。"]
        for o in self.out_sheets:
            rows = o.max_dest_row() + 1
            cols = o.max_dest_col() + 1
            lines.append(f"  出力シート「{o.name}」: {len(o.blocks)} ブロック / {rows} 行 × {cols} 列")
        for e in self.errors:
            loop = f"({e['loop']})" if e.get("loop") else ""
            lines.append(f"  ⚠ {e['n']}行目{loop}: {e['msg']} — {e['text']}")
        return lines


# --- 繰り返しの展開(元実装の parseValueList / expandScript) ---


def parse_value_list(text: str, kind: str) -> list[str]:
    """「10, 12, 17, 20」「10-14」「B-E」を値の並びにする。

    英字の範囲(B-E)を広げるのは kind が「列」のときだけ。
    (シート名に含まれるハイフンを巻き込まないため)
    """
    out: list[str] = []
    for part in re.split(r"[,、]", to_half(text)):
        p = part.strip()
        if not p:
            continue
        m = re.fullmatch(r"(\d{1,7})\s*[-–~〜]\s*(\d{1,7})", p)
        if m:
            a, b = int(m.group(1)), int(m.group(2))
            step = 1 if a <= b else -1
            out.extend(str(v) for v in range(a, b + step, step))
            continue
        if kind == "列":
            mc = re.fullmatch(r"([A-Za-z]{1,3})\s*[-–~〜]\s*([A-Za-z]{1,3})", p)
            if mc:
                ca, cb = col_index(mc.group(1)), col_index(mc.group(2))
                if ca >= 0 and cb >= 0:
                    step = 1 if ca <= cb else -1
                    out.extend(col_name(c) for c in range(ca, cb + step, step))
                    continue
        out.append(p.upper() if kind == "列" and re.fullmatch(r"[A-Za-z]{1,3}", p) else p)
    return out


def subst_vars(text: str, name: str, value: str) -> str:
    return str(text).replace("{" + name + "}", value).replace("｛" + name + "｝", value)


RE_REPEAT = re.compile(r"^(?:繰り返し|くりかえし|repeat)\s+([^\s=]+)\s*=\s*(.+)$", re.I)
RE_REPEAT_HEAD = re.compile(r"^(?:繰り返し|くりかえし|repeat)\s", re.I)
RE_REPEAT_END = re.compile(r"^(?:ここまで|終わり|おわり|end)$", re.I)


@dataclass
class Step:
    n: int  # 手順書の行番号(1 始まり)
    text: str
    loop: str | None = None
    it: int | None = None  # 繰り返しの何周目か


def expand_script(text: str) -> tuple[list[Step], list[dict]]:
    """手順書を実行できる行の並びに展開する。

    「繰り返し 行 = 10, 12」〜「ここまで」を値の数だけ展開し、{行} を差し替える。
    入れ子の繰り返しには対応していない(元の実装と同じ)。
    """
    lines = re.sub(r"\r\n?", "\n", str(text)).split("\n")
    steps: list[Step] = []
    errors: list[dict] = []
    i = 0
    while i < len(lines):
        t = to_half(lines[i]).strip()
        m = RE_REPEAT.match(t)
        if not m:
            steps.append(Step(n=i + 1, text=lines[i]))
            i += 1
            continue

        name = re.sub(r"^[{｛]|[}｝]$", "", m.group(1))
        values = parse_value_list(m.group(2), name)
        if not values:
            errors.append({"n": i + 1, "msg": "繰り返しの値を読み取れません"})
            i += 1
            continue

        body: list[Step] = []
        j = i + 1
        closed = False
        while j < len(lines):
            bt = to_half(lines[j]).strip()
            if RE_REPEAT_HEAD.match(bt):
                errors.append({"n": j + 1, "msg": "繰り返しの入れ子には対応していません"})
                break
            if RE_REPEAT_END.match(bt):
                closed = True
                break
            body.append(Step(n=j + 1, text=lines[j]))
            j += 1
        if not closed:
            errors.append({"n": i + 1, "msg": "「ここまで」が見つかりません"})
        if not body:
            errors.append({"n": i + 1, "msg": "繰り返しの中身が空です"})

        for vi, v in enumerate(values):
            for b in body:
                steps.append(
                    Step(n=b.n, text=subst_vars(b.text, name, v), loop=f"{name}={v}", it=vi)
                )
        i = j + 1
    return steps, errors


# --- 1 行の実行(元実装の execLine) ---

RE_SKIP_META = re.compile(r"^(説明|メモ|備考|作成者|作成日|元ファイル)\s*[:：]")
RE_GENERATE = re.compile(r"^(?:生成|書き出し?|ダウンロード|generate|export)$", re.I)
RE_ADD_SHEET = re.compile(
    r"^(?:シート追加|新規シート|シートを追加|出力シート|add\s*sheet)\s+(.+)$", re.I
)
RE_TRANS = re.compile(r"(転置|行列を?入れ替)")
RE_TRANS_STRIP = re.compile(r"[（(]?\s*(?:転置|行列を?入れ替える?)\s*[）)]?")
RE_CONT_RIGHT = re.compile(r"(?:右|横)\s*に\s*(?:続けて|並べて)?\s*(?:置く|配置|入れる|並べる)\s*$")
RE_CONT_DOWN1 = re.compile(r"(?:(?:その)?(?:下|後ろ|次の行)\s*に\s*)?続けて\s*(?:置く|配置|入れる)?\s*$")
RE_CONT_DOWN2 = re.compile(r"(?:その)?(?:下|後ろ)\s*に\s*(?:置く|貼り?付ける?|配置)\s*$")
RE_ARROW = re.compile(r"\s*(?:->|=>|→|⇒)\s*")
RE_JP_PLACE = re.compile(r"^(.*?)を(.*?)(?:に|へ)(?:置|貼|配置|コピー|入れ|移)")
RE_JP_PLACE2 = re.compile(r"^(.*?)(?:を)?\s*(?:に|へ)?\s*(.*?)(?:に|へ)(?:置|貼|配置|コピー)")
RE_CELL_IN = re.compile(r"([A-Z]{1,3}\d{1,7})", re.I)


class Interpreter:
    """手順書を実行して出力シートを組み立てる。"""

    def __init__(self, sheets: list[SourceSheet]):
        self.sheets = sheets
        self.out: list[OutSheet] = []
        self.last: tuple[Block, OutSheet] | None = None
        self.flow_col: int | None = None
        self.shift: dict[str, int] | None = None
        self.generate = False

    # -- シート名の解決 --

    def match_sheet(self, text: str) -> int | None:
        """文章の中から元データのシートを見つける。長い名前を優先。

        「[売上_2026.xlsx]売上明細」のようにファイル名で絞り込める(同名シート対策)。
        """
        only: str | None = None
        fm = re.search(r"[\[［]([^\]］]+)[\]］]", text)
        if fm:
            fname = fm.group(1).strip()
            names = {s.file for s in self.sheets}
            if fname in names:
                only = fname
            else:
                for n in names:
                    if fname in n:
                        only = n
                        break
        best: tuple[int, int] | None = None  # (シート番号, 名前の長さ)
        for i, sh in enumerate(self.sheets):
            if only is not None and sh.file != only:
                continue
            if sh.name in text and (best is None or len(sh.name) > best[1]):
                best = (i, len(sh.name))
        return best[0] if best else None

    def match_out(self, text: str) -> OutSheet | None:
        best: tuple[OutSheet, int] | None = None
        for o in self.out:
            if o.name in text and (best is None or len(o.name) > best[1]):
                best = (o, len(o.name))
        return best[0] if best else None

    def use_or_add_out(self, name: str) -> OutSheet:
        """同じ名前の出力シートがあれば使い回す(何度実行しても増えない)。"""
        for o in self.out:
            if o.name == name:
                return o
        o = OutSheet(name=name)
        self.out.append(o)
        return o

    # -- 範囲の解釈 --

    def parse_range(self, text: str, sheet_idx: int) -> Range | None:
        sh = self.sheets[sheet_idx]
        last_row = max(sh.used_rows - 1, 0)
        last_col = max(min(sh.used_cols, MAX_COLS) - 1, 0)
        t = re.sub(r"\s+", "", text)

        if re.search(r"全体|すべて|全部|ぜんぶ|whole|all", t, re.I):
            return Range(0, 0, last_row, last_col)

        m = re.search(
            r"([A-Z]{1,3}\d{1,7})\s*(?::|から|〜|~|-|–|—|to)\s*([A-Z]{1,3}\d{1,7})", t, re.I
        )
        if m:
            a, b = parse_cell(m.group(1)), parse_cell(m.group(2))
            if a and b:
                return norm_range(a[0], a[1], b[0], b[1])

        m = re.search(r"(\d{1,7})\s*行目?\s*(?::|から|〜|~|-|to)\s*(\d{1,7})\s*行目?", t, re.I)
        if m:
            return Range(int(m.group(1)) - 1, 0, int(m.group(2)) - 1, last_col)

        m = re.search(r"(\d{1,7})\s*行目", t)
        if m and not re.search(r"[A-Z]\d", t, re.I):
            r = int(m.group(1)) - 1
            return Range(r, 0, r, last_col)

        m = re.search(r"([A-Z]{1,3})\s*列\s*(?::|から|〜|~|-|to)\s*([A-Z]{1,3})\s*列?", t, re.I)
        if m:
            c1, c2 = col_index(m.group(1)), col_index(m.group(2))
            if c1 >= 0 and c2 >= 0:
                return norm_range(0, c1, last_row, c2)

        m = re.search(r"([A-Z]{1,3})\s*列", t, re.I)
        if m:
            c = col_index(m.group(1))
            if c >= 0:
                return Range(0, c, last_row, c)

        m = re.search(r"([A-Z]{1,3}\d{1,7})", t, re.I)
        if m:
            p = parse_cell(m.group(1))
            if p:
                return Range(p[0], p[1], p[0], p[1])
        return None

    # -- 1 行の実行 --

    def exec_line(self, raw: str) -> dict:
        """戻り値: {"skip": True} | {"ok": True, "msg": …} | {"err": 理由}"""
        t = to_half(raw).strip()
        if not t or t.startswith("#"):
            return {"skip": True}
        if RE_SKIP_META.match(t):
            return {"skip": True}

        if RE_GENERATE.match(t):
            self.generate = True
            return {"ok": True, "msg": "書き出し"}

        m = RE_ADD_SHEET.match(t)
        if m:
            o = self.use_or_add_out(m.group(1).strip())
            self.last = None
            return {"ok": True, "msg": f"出力シート「{o.name}」"}

        if not self.sheets:
            return {"err": "元データが読み込まれていません"}

        # 「(転置)」の指定を取り出す
        trans = bool(RE_TRANS.search(t))
        t = RE_TRANS_STRIP.sub("", t).strip()

        # 「続けて置く」= 直前に置いたブロックの下(または右)に続ける
        cont = None
        mc = RE_CONT_RIGHT.search(t)
        if mc:
            cont = "right"
        else:
            mc = RE_CONT_DOWN1.search(t)
            if mc:
                cont = "down"
            else:
                mc = RE_CONT_DOWN2.search(t)
                if mc:
                    cont = "down"
        if cont:
            t = re.sub(r"[をは]\s*$", "", t[: mc.start()]).strip()

        # 「XXX を YYY に置く」/「XXX → YYY」に分解
        src_txt, dst_txt = t, ""
        arrow = RE_ARROW.search(t)
        if arrow:
            src_txt = t[: arrow.start()]
            dst_txt = t[arrow.end() :]
        elif not cont:
            jm = RE_JP_PLACE.match(t)
            if jm:
                src_txt, dst_txt = jm.group(1), jm.group(2)
            else:
                jm2 = RE_JP_PLACE2.match(t)
                if jm2:
                    src_txt, dst_txt = jm2.group(1), jm2.group(2)

        # 出力先シート
        o = self.match_out(dst_txt)
        dst_clean = dst_txt.replace(o.name, "") if o else dst_txt
        if o is None and cont and self.last:
            o = self.last[1]
        if o is None:
            o = self.out[-1] if self.out else self.use_or_add_out("抜粋1")

        anchor: tuple[int, int] | None = None
        am = RE_CELL_IN.search(re.sub(r"\s+", "", dst_clean))
        if am:
            anchor = parse_cell(am.group(1))
        # 繰り返しの 2 周目以降は、位置を書いた行も前の周の下へ送る
        if anchor and self.shift and o.name in self.shift:
            anchor = (anchor[0] + self.shift[o.name], anchor[1])

        # 元データのシート
        sheet_idx = self.match_sheet(src_txt)
        if sheet_idx is None:
            nm = re.match(r"^\s*(.+?)\s*[の!]", src_txt)
            if nm and nm.group(1).strip():
                return {"err": f"シート「{nm.group(1).strip()}」が見つかりません"}
            return {"err": "元データのシートを特定できません"}
        src_clean = src_txt.replace(self.sheets[sheet_idx].name, "")
        g = self.parse_range(src_clean, sheet_idx)
        if not g:
            return {"err": "範囲を読み取れません"}

        # 置き場所を決める
        if anchor:
            self.flow_col = anchor[1]
        elif cont and self.last:
            lb = self.last[0]
            if cont == "right":
                anchor = (lb.dr, lb.dc + lb.width)
            else:
                # 下に続ける = 使用済みの最終行の次の行へ。列は行の左端に戻す
                col = self.flow_col if self.flow_col is not None else lb.dc
                anchor = (o.max_dest_row() + 1, col)
                self.flow_col = col
        if not anchor:
            # 空いている場所(最下段の 2 行下)
            m_row = o.max_dest_row()
            anchor = (0 if m_row < 0 else m_row + 2, 0)
            self.flow_col = anchor[1]

        block = Block(sheet=sheet_idx, src=g, dr=max(0, anchor[0]), dc=max(0, anchor[1]), trans=trans)
        o.blocks.append(block)
        self.last = (block, o)
        return {
            "ok": True,
            "msg": f"{self.sheets[sheet_idx].name}!{g.label()} → {o.name}!{a1(block.dr, block.dc)}"
            + ("(転置)" if trans else ""),
        }


def run_script(text: str, sheets: list[SourceSheet]) -> RunResult:
    """手順書を実行して出力シートを組み立てる。"""
    steps, errors = expand_script(text)
    if errors:
        first = errors[0]
        raise ProcedureError(f"手順書に問題があります: {first['n']}行目 {first['msg']}")

    interp = Interpreter(sheets)
    result = RunResult(out_sheets=interp.out)
    cur_iter = -1
    for st in steps:
        # 繰り返しの周が変わったら、位置指定をずらす基準を取り直す
        if st.it is None:
            interp.shift = None
            cur_iter = -1
        elif st.it != cur_iter:
            cur_iter = st.it
            interp.shift = {}
            if cur_iter > 0:
                for o in interp.out:
                    interp.shift[o.name] = o.max_dest_row() + 1
        try:
            r = interp.exec_line(st.text)
        except Exception as e:  # 1 行の失敗で全体を止めない
            r = {"err": str(e)}
        if r.get("skip"):
            continue
        if r.get("err"):
            result.errors.append(
                {"n": st.n, "msg": r["err"], "text": to_half(st.text).strip(), "loop": st.loop}
            )
        else:
            result.applied += 1
            if r.get("msg"):
                result.messages.append(r["msg"])

    if not interp.out:
        interp.use_or_add_out("抜粋1")
    result.out_sheets = interp.out
    result.generate = interp.generate
    return result


def write_output(result: RunResult, sheets: list[SourceSheet], output_path: str | Path) -> Path:
    """組み立てた出力シートを Excel として書き出す。"""
    output_path = Path(output_path)
    workbook = Workbook()
    workbook.remove(workbook.active)
    for o in result.out_sheets:
        # Excel のシート名は 31 文字まで／一部の記号が使えない
        title = re.sub(r"[\[\]\*/\\?:]", "_", o.name)[:31] or "抜粋1"
        ws = workbook.create_sheet(title)
        for b in o.blocks:
            src = sheets[b.sheet]
            for i in range(b.src.height):
                for j in range(b.src.width):

"""コアプログラムを標準出力キャプチャ付きで実行する。

Jupyter Lab ではなくターミナルで `python` を実行したときと同じ print 出力を、

コアの実体は 2 通り:

1. **Dify に貼り付けたコード**(推奨)
   社内のコア Python を外(このリポジトリ)に出せないため、Dify のワークフローの
   環境変数に丸ごと貼り付けておき、実行時に文字列として受け取ってここで実行する。
   コードは Dify(プラグインのプロセス内)だけで実行され、外部には送信されない。
2. **同梱コード**(plugin/core_logic/core/generate_plan.py)
   貼り付けが無い場合の既定。現在は動作確認用のモック実装。
"""

from __future__ import annotations

import contextlib
import inspect
import io
import traceback
import types
from dataclasses import dataclass, field

from .core import generate_plan


@dataclass
class CoreResult:
    success: bool
    log: str
    output_files: list[Path] = field(default_factory=list)


def run_core(
    bulk_xlsx_path: str | Path,
    output_dir: str | Path,
    fiscal_year: int,
    plan_fiscal_year: int | None = None,
    start_month: int = 4,
    core_code: str | None = None,
) -> CoreResult:
    """コアプログラムを実行し、print 出力(stdout/stderr)と出力ファイルを回収する。

    Args:
        fiscal_year: 実績(入力)の年度
        plan_fiscal_year: 生成する資金計画の年度(省略時は実績年度の翌年度)
        start_month: 年度の開始月(3月決算なので既定は 4)
        core_code: Dify 側に貼り付けたコア Python の本文。指定があれば同梱コードの
            代わりにこれを実行する(空なら同梱コードにフォールバック)

    コア側で例外が起きても送出せず、それまでの print 出力+トレースバックを
    ログとして返す(Dify の画面にそのまま表示するため)。
    """
    plan_fiscal_year = plan_fiscal_year or fiscal_year + 1
    if core_code and str(core_code).strip():
        return _run_injected(
            str(core_code), bulk_xlsx_path, output_dir, fiscal_year, plan_fiscal_year, start_month
        )

    buffer = io.StringIO()
    try:
        with contextlib.redirect_stdout(buffer), contextlib.redirect_stderr(buffer):
            outputs = generate_plan.main(
                input_path=Path(bulk_xlsx_path),
                output_dir=Path(output_dir),
                fiscal_year=fiscal_year,
                plan_fiscal_year=plan_fiscal_year,
                start_month=start_month,
            )
        return _collect(buffer, outputs, None)
    except Exception:
        log = buffer.getvalue() + "\nコアプログラムでエラーが発生しました:\n" + traceback.format_exc()
        return CoreResult(success=False, log=log)


def _run_injected(
    code_text: str,
    bulk_xlsx_path: str | Path,
    output_dir: str | Path,
    fiscal_year: int,
    plan_fiscal_year: int,
    start_month: int,
) -> CoreResult:
    """Dify 側に貼り付けられたコア Python を実行する。

    貼り付けたコードの書き方は 2 通りどちらでもよい:

    - **そのまま実行する書き方**(Jupyter のセルを並べたようなコード)
      `input_path` / `output_dir` / `fiscal_year` / `plan_fiscal_year` / `start_month`
      が変数として最初から使える。出力ファイルは output_dir に保存すれば自動で回収される。
    - **main() を定義する書き方**
      `def main(input_path, output_dir, fiscal_year, ...)` を定義すると、コードの実行後に
      その main() が呼ばれる(引数は定義にあるものだけ渡す)。出力ファイルのパスの
      リストを return してもよいし、return しなければ output_dir から自動で回収する。
    """

def execute_code(
    code_text: str,
    variables: dict,
    output_dir: str | Path,
    error_label: str = "貼り付けた Python",
) -> CoreResult:
    """貼り付けられた Python を、変数を与えて実行する共通処理。

    - variables はコードの中で最初から使えるグローバル変数になる
    - モジュールとして実行するため __name__ は "__main__" ではなく、
      `if __name__ == "__main__":` のブロックは実行されない(main() は下で呼ぶ)
    - def main(...) が定義されていれば、定義にある引数だけを渡して呼ぶ
    - 出力ファイルは main の戻り値、無ければ output_dir から自動回収する
    """
    buffer = io.StringIO()
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)
    module = types.ModuleType("core_injected")
    module.__dict__.update(variables)
    try:
        compiled = compile(code_text, f"<{error_label}>", "exec")
        with contextlib.redirect_stdout(buffer), contextlib.redirect_stderr(buffer):
            exec(compiled, module.__dict__)
            main = module.__dict__.get("main")
            outputs = main(**_accepted_kwargs(main, variables)) if callable(main) else None
        return _collect(buffer, outputs, output_dir)
    except Exception:
        log = buffer.getvalue() + f"\n{error_label} でエラーが発生しました:\n" + traceback.format_exc()
        return CoreResult(success=False, log=log)


def _accepted_kwargs(func, params: dict) -> dict:
    """main() の定義にある引数だけを渡す(貼り付け側の書き方の自由度のため)。"""
    try:
        sig = inspect.signature(func)
    except (TypeError, ValueError):
        return {}
    if any(p.kind is inspect.Parameter.VAR_KEYWORD for p in sig.parameters.values()):
        return dict(params)
    return {name: value for name, value in params.items() if name in sig.parameters}


def _collect(buffer: io.StringIO, outputs, scan_dir: Path | None) -> CoreResult:
    """出力ファイルを確定する。outputs(main の戻り値)が無ければ scan_dir から拾う。"""
    if outputs is None and scan_dir is not None:
        files = sorted(p for p in scan_dir.rglob("*") if p.is_file())
    else:
        files = [Path(p) for p in outputs or []]
        missing = [p for p in files if not p.exists()]
        if missing:
            names = ", ".join(p.name for p in missing)
            return CoreResult(
                success=False,
                log=buffer.getvalue() + f"\nエラー: 出力ファイルが見つかりません: {names}",
            )
    log = buffer.getvalue()
    if not files:
        log += "\n注意: 出力ファイルが作られませんでした(output_dir に保存してください)。"
    return CoreResult(success=True, log=log, output_files=files)

def format_sheet_name(pattern: str, year: int, month: int) -> str:
    """シート名のパターンを、暦年と月から実際のシート名に変換する。

    使えるプレースホルダ:
        {R}  令和年(1〜)      {RR} 令和年2桁      例: 令和8年 → 8 / 08
        {M}  月(1〜12)        {MM} 月2桁          例: 4月 → 4 / 04
        {Y}  西暦4桁          {YY} 西暦下2桁      例: 2026 / 26

    例) "R{R}{MM}" + 令和8年11月 → "R811"
    """
    reiwa = year - 2018  # 令和1年 = 2019年
    return (
        pattern.replace("{RR}", f"{reiwa:02d}")
        .replace("{R}", str(reiwa))
        .replace("{MM}", f"{month:02d}")
        .replace("{M}", str(month))
        .replace("{YY}", f"{year % 100:02d}")
        .replace("{Y}", str(year))
    )


def _generate_months(pattern: object, fiscal_year: int, start_month: int, errors: list[str]) -> list[MonthSpec]:
    """sheet_pattern から 12 ヶ月分のシート名を生成する。"""

        )
        return []
    pattern = pattern.strip()

    months: list[MonthSpec] = []

            raise PreprocessError("数値として読み取れないセルがあります:\n- " + "\n- ".join(shown))

                f"注意: 空欄セル {result.blank_cells} 件を 0 として扱いました。"
                "数式のセルが空欄扱いになる場合は、元データを一度 Excel で開いて保存し直してください"
                "(数式の計算結果が保存されていないと値を読み取れません)。"
            )
        return result
    finally:
        workbook.close()


def _to_amount(value: object) -> tuple[float, bool, str | None]:
    """セル値を金額(float)に変換する。戻り値は (値, 空欄か, エラーメッセージ)。"""
    if value is None:
        return 0.0, True, None
    if isinstance(value, bool):
        return 0.0, False, f"TRUE/FALSE が入っています({value})"
    if isinstance(value, (int, float)):
        return float(value), False, None
    if isinstance(value, str):
        text = value.strip().replace(",", "").replace("¥", "").replace("\\", "")
        if not text:
            return 0.0, True, None
        try:
            return float(text), False, None
        except ValueError:
            return 0.0, False, f"数値ではない文字列が入っています(「{value}」)"
    return 0.0, False, f"数値として扱えない値が入っています({value!r})"

コア Python の渡し方は 2 通りあります。**社内のコードを外(このリポジトリ)に
出せない場合は A の貼り付け方式**を使ってください。こちらが標準の運用です。

## A. Dify に貼り付ける(推奨・コードはリポジトリに置かない)

コードは Dify のワークフローの**環境変数 `CORE_CODE`** に丸ごと貼り付けます。

1. Dify でワークフローの編集画面を開く
2. 「環境変数(ENV)」パネルで `CORE_CODE`(string)を開く
3. ノートブックのコードを**そのまま全部貼り付けて**保存 → 公開し直す

実行のたびにプラグインが `CORE_CODE` の中身を受け取り、プラグインのプロセス内で
実行します。**コードが Dify の外へ送信されることはありません。**
空のままの場合は、このフォルダの `generate_plan.py`(モック実装)が動きます。

### 貼り付けるコードの書き方

Jupyter のセルを並べたような「そのまま実行する」書き方で OK です。
次の変数が最初から使えます。

| 変数 | 内容 |
|------|------|
| `input_path` | 入力用一括 Excel のパス(前処理が生成したもの) |
| `output_dir` | 出力ファイルの保存先(`Path`。ここに保存すれば自動で回収される) |
| `fiscal_year` | 実績データの年度(例: 2025 = 令和7年度) |
| `plan_fiscal_year` | 作成する計画の年度(実績の翌年度) |
| `start_month` | 年度の開始月(3月決算なので 4) |
